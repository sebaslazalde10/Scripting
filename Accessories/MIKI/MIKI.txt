(defun c:MIKI2PSUMACC (/)

  ;; Inicializar entorno Byrne
  (ByrneStart "MIKI_2PORTS_UM" "30")

  ;; =========================
  ;; INSERTAR ACCESORIO
  ;; =========================

  (prompt "\nInserta y rota MIKI 2 PORTS UM: ")

  (ByrnePurgeBlock "MIKI_2PORTS_UM")

  (ByrneInsertBlock
    "MIKI_2PORTS_UM"
    pause
  )

  ;; =========================
  ;; BOM
  ;; =========================

  (ByrneAddBOM "COMMERCIAL NAME" 1)

  ;; =========================
  ;; RESTAURAR ENTORNO
  ;; =========================

  (ByrneEnd)

  (princ "\n COMMERCIAL SIMPLIFIED NAME insertado correctamente.")
  (princ)

)
